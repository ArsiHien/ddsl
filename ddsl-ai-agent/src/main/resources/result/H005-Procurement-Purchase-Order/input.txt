BoundedContext Procurement { domain { Aggregate PurchaseOrder { purchaseOrderId uuid identity; supplierId text required; requesterId text required; lines list of PurchaseOrderLine; totalAmount Money; status text default DRAFT; approvedAt datetime optional. PurchaseOrderLine entity with lineId uuid identity, itemCode text required, quantity int min 1, unitCost Money required. operations { when submitting purchase order: supplier requester and lines are required; calculate total from line costs; status SUBMITTED; emit PurchaseOrderSubmitted. when approving purchase order with approvedAt and approverId: status must be SUBMITTED, approverId required, totalAmount greater than zero; set status APPROVED and approvedAt; emit PurchaseOrderApproved. } } ValueObject Money amount decimal at least zero currency text required max 3. DomainService ApprovalPolicy { when checking approval requirement with totalAmount as Money and requesterId as String: then calculate approvalRequired as totalAmount is greater than 1000. } } events { PurchaseOrderSubmitted id uuid submittedAt datetime total Money. PurchaseOrderApproved id uuid approverId string approvedAt datetime. } factories { PurchaseOrderFactory creates PurchaseOrder from supplierId requesterId and lines requiring lines not empty and returning created order. } }
